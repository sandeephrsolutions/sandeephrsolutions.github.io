rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /candidates/{userId} {
      allow read, create, update, delete:
        if request.auth != null
        && request.auth.uid == userId;
    }

    // Public jobs contain only information intended for candidates.
    // Company name, contact person, mobile and email are never written here
    // when a job comes from an employer request.
    match /jobs/{jobId} {
      allow read: if true;

      allow create, update, delete:
        if request.auth != null
        && request.auth.uid == "A5y0cl3ZvRdlzFgDuQsYUjVCeNq2";
    }

    match /applications/{applicationId} {
      allow create:
        if request.auth != null;

      allow read, update, delete:
        if request.auth != null
        && request.auth.uid == "A5y0cl3ZvRdlzFgDuQsYUjVCeNq2";
    }

    // Employer submissions may be created from the public Employer page.
    // Only the admin can read/change/delete them.
    match /jobRequests/{requestId} {
      allow create: if true;

      allow read, update, delete:
        if request.auth != null
        && request.auth.uid == "A5y0cl3ZvRdlzFgDuQsYUjVCeNq2";
    }

    // Existing employer collection, kept for compatibility.
    match /employers/{employerId} {
      allow create: if true;

      allow read, update, delete:
        if request.auth != null
        && request.auth.uid == "A5y0cl3ZvRdlzFgDuQsYUjVCeNq2";
    }
  }
}
